// ---- 常量 ----------------------------------------------------------------
const uint MAT_EMPTY = 0u;
const uint MAT_WATER = 6u;
const float WATER_ETA = 1.33;                     // 水的折射率
const vec3  WATER_SIGMA = vec3(0.55, 0.18, 0.10); // 水的吸收系数 /m（红衰减最快 → 青绿色调）
const float WATER_F0 = 0.02;
const float GLASS_F0 = 0.04;
const uint CAUSTICS_RES = 256u;
const float PI = 3.14159265359;
const float INF = 1e30;

// 球体材质类型（albedo.w）
const float SMAT_DIFFUSE = 0.0;
const float SMAT_METAL   = 1.0;
const float SMAT_GLASS   = 2.0;

// 材质类型（matEmission.w）
const float MTYPE_DIFFUSE  = 0.0;
const float MTYPE_METAL    = 1.0;
const float MTYPE_EMISSIVE = 2.0;
const float MTYPE_WATER    = 3.0;
const uint MAT_POOLTILE = 4u;

// ---- 工具 ----------------------------------------------------------------
ivec3 worldSize() { return ivec3(int(P.uPoolWorld.z), int(P.uPoolWorld.w), int(P.uMisc.x)); }
vec3 safeDir(vec3 d) { return mix(d, vec3(1e-10), lessThan(abs(d), vec3(1e-10))); }

// 程序化水面波纹法线：迭代分形波（多倍频方向旋转 + 时间流动）。
// 借鉴 yaohai-epic 水面着色器的迭代言波思路：exp(sin-1) 非对称波形，
// 每级频率/速度递增、权重递减、方向旋转——此处取解析梯度做法线，
// 幅度控制在 ~0.2 坡度以内（保证水下仰视的 Snell window 开放）。
vec3 waterNormal(vec2 p) {
    float t = P.uMisc.z;             // 流动时间（0 = 静止）
    vec2 pos = p * 0.55;
    float iter = 0.0;
    float frequency = 2.1;
    float speed = 1.3;
    float weight = 1.0;
    float totalW = 0.0;
    vec2 grad = vec2(0.0);
    for (int i = 0; i < 5; i++) {
        vec2 dir = vec2(sin(iter), cos(iter));
        float phase = dot(dir, pos) * frequency + t * speed;
        float w = exp(sin(phase) - 1.0);      // 非对称波高
        grad += w * cos(phase) * weight * dir * frequency;   // 解析梯度
        iter += 1.9;                          // 每级转向 ~17°
        weight *= 0.72;
        frequency *= 1.55;
        speed *= 1.12;
        totalW += weight;
    }
    float amp = 0.16 / max(totalW, 1e-3);     // 归一化坡度幅度
    return normalize(vec3(-grad.x * amp, 1.0, -grad.y * amp));
}

uint pcgHash(inout uint s) {
    s = s * 747796405u + 2891336453u;
    uint w = ((s >> ((s >> 28u) + 4u)) ^ s) * 277803737u;
    return (w >> 22u) ^ w;
}
float rnd(inout uint s)  { return float(pcgHash(s)) * (1.0 / 4294967296.0); }
vec2  rnd2(inout uint s) { return vec2(rnd(s), rnd(s)); }

// 立方体哈希（标准公开哈希，程序化纹理/坏灯/星星/云共用）
float hash13(vec3 p) {
    p = fract(p * 0.1031);
    p += dot(p, p.zyx + 31.32);
    return fract((p.x + p.y) * p.z);
}

uint voxelAt(ivec3 c) {
    ivec3 ws = worldSize();
    if (any(lessThan(c, ivec3(0))) || any(greaterThanEqual(c, ws))) return MAT_EMPTY;
    int idx = c.x + c.y * ws.x + c.z * ws.x * ws.y;
    return (voxelWords[idx >> 2] >> ((idx & 3) * 8)) & 0xFFu;
}
vec4 matAlbedo(uint id)   { return matData[int(id) * 2 + 0]; }
vec4 matEmission(uint id) { return matData[int(id) * 2 + 1]; } // rgb = 自发光, w = 类型

// GI 缓存读写（放这里是因为依赖 worldSize/matAlbedo）
vec3 cacheE(vec3 p) {
    return texture(giCacheSmp, p / vec3(worldSize())).rgb;
}
void cacheSplat(vec3 p, vec3 Etot, float blend) {
    ivec3 c = clamp(ivec3(p * 0.5), ivec3(0), ivec3(15));
    vec4 old = imageLoad(giCacheImg, c);
    imageStore(giCacheImg, c, vec4(mix(old.rgb, Etot, blend), 1.0));
}

// ---- 统一介质（空气/水/彩色玻璃体素）------------------------------------
// 材质 ID 10..16 = 彩色玻璃体素（材质表 albedo.rgb 存吸收系数 σ）
const uint MAT_GLASS_MIN = 10u;
const uint MAT_GLASS_MAX = 16u;
bool isGlassVox(uint m)  { return m >= MAT_GLASS_MIN && m <= MAT_GLASS_MAX; }
bool isMedium(uint m)    { return m == MAT_WATER || isGlassVox(m); }
float mediumEta(uint m)  { return m == MAT_WATER ? WATER_ETA : 1.5; } // m!=0 时调用
vec3 mediumSigma(uint m) {
    if (m == MAT_WATER) return WATER_SIGMA;
    if (isGlassVox(m)) return matAlbedo(m).rgb;
    return vec3(0.0);
}

// 精确介质菲涅尔（s/p 偏振平均；全内反射返回 1）——与原版同款物理公式
float fresnelDielectric(float cosI, float etaI, float etaT) {
    if (etaI == etaT) return 0.0;
    float c = clamp(cosI, 0.0, 1.0);
    float s2t = (etaI / etaT) * (etaI / etaT) * (1.0 - c * c);
    if (s2t >= 1.0) return 1.0;              // TIR
    float ct = sqrt(1.0 - s2t);
    float rs = (etaI * c - etaT * ct) / (etaI * c + etaT * ct);
    float rp = (etaT * c - etaI * ct) / (etaT * c + etaI * ct);
    return 0.5 * (rs * rs + rp * rp);
}

// 与世界 AABB 求交；返回 t 区间
bool worldEnter(vec3 ro, vec3 rd, out float tEnter, out float tExit) {
    vec3 invD = 1.0 / safeDir(rd);
    vec3 tA = (vec3(0.0) - ro) * invD;
    vec3 tB = (vec3(worldSize()) - ro) * invD;
    vec3 mn = min(tA, tB), mx = max(tA, tB);
    tEnter = max(max(mn.x, mn.y), mn.z);
    tExit  = min(min(mx.x, mx.y), mx.z);
    return tExit >= tEnter && tExit > 0.0;
}

// ---- 球体相交 ------------------------------------------------------------
float sphereHit(vec3 ro, vec3 rd, vec4 s, float tmax) {
    vec3 oc = ro - s.xyz;
    float b = dot(oc, rd);
    float c = dot(oc, oc) - s.w * s.w;
    float h = b * b - c;
    if (h < 0.0) return -1.0;
    h = sqrt(h);
    float t = -b - h;
    if (t > 1e-4 && t < tmax) return t;
    t = -b + h;
    if (t > 1e-4 && t < tmax) return t;
    return -1.0;
}
float sphereExitT(vec3 ro, vec3 rd, vec4 s) {
    vec3 oc = ro - s.xyz;
    float b = dot(oc, rd);
    float c = dot(oc, oc) - s.w * s.w;
    return -b + sqrt(max(b * b - c, 0.0));
}
int nearestSphere(vec3 ro, vec3 rd, float tmax, out float tHit) {
    int best = -1; tHit = tmax;
    int n = int(P.uMisc.w);
    for (int i = 0; i < n; i++) {
        float t = sphereHit(ro, rd, sphereData[i * 3], tHit);
        if (t > 0.0) { tHit = t; best = i; }
    }
    return best;
}
// 球体阴影透过率：不透明球归零；玻璃球按 Beer-Lambert 染色（彩色投影）
vec3 sphereShadowTint(vec3 ro, vec3 rd, float dist) {
    vec3 T = vec3(1.0);
    int n = int(P.uMisc.w);
    for (int i = 0; i < n; i++) {
        float t = sphereHit(ro, rd, sphereData[i * 3], dist);
        if (t <= 0.0) continue;
        float mtype = sphereData[i * 3 + 1].w;
        if (mtype == SMAT_DIFFUSE || mtype == SMAT_METAL) return vec3(0.0);
        if (mtype == SMAT_GLASS) {
            float chord = 2.0 * sphereData[i * 3].w * 0.75;  // 平均弦长近似
            T *= exp(-sphereData[i * 3 + 1].rgb * chord);
        }
    }
    return T;
}

// ---- 太阳镜面高光（specular NEE）------------------------------------------
// 视线关于法线的镜面方向落在太阳角半径内 → 返回太阳圆盘辐射亮度。
// 天空不含日盘：漫反射走 NEE，镜面走本函数，避免重复计数。
const float SUN_COS_LIMIT = 0.99988; // ≈0.81° 角半径
vec3 sunSpecular(vec3 rd, vec3 n) {
    if (P.uFrame.w < 0.5 || P.uSun.y < 0.0) return vec3(0.0);   // 夜间无太阳高光
    vec3 r = reflect(rd, n);
    if (dot(r, P.uSun.xyz) <= SUN_COS_LIMIT) return vec3(0.0);
    return P.uSunColor.rgb * (P.uSun.w * 80.0);
}

// 水下命中点 → 焦散辐照度（双线性解码定点光子图）
vec3 causticsIrradiance(vec3 p) {
    if (p.y > P.uSkyWater.y) return vec3(0.0);
    vec2 poolMin = P.uSkyWater.zw, poolSize = P.uPoolWorld.xy;
    vec2 uv = (p.xz - poolMin) / poolSize;
    if (any(lessThan(uv, vec2(0.0))) || any(greaterThan(uv, vec2(1.0)))) return vec3(0.0);
    vec2 g = uv * float(CAUSTICS_RES) - 0.5;
    vec2 f = fract(g);
    ivec2 c0 = clamp(ivec2(floor(g)), ivec2(0), ivec2(int(CAUSTICS_RES) - 1));
    float e00 = float(imageLoad(causticsImg, clamp(c0, ivec2(0), ivec2(int(CAUSTICS_RES)-1))).r);
    float e10 = float(imageLoad(causticsImg, clamp(c0 + ivec2(1, 0), ivec2(0), ivec2(int(CAUSTICS_RES)-1))).r);
    float e01 = float(imageLoad(causticsImg, clamp(c0 + ivec2(0, 1), ivec2(0), ivec2(int(CAUSTICS_RES)-1))).r);
    float e11 = float(imageLoad(causticsImg, clamp(c0 + ivec2(1, 1), ivec2(0), ivec2(int(CAUSTICS_RES)-1))).r);
    float raw = mix(mix(e00, e10, f.x), mix(e01, e11, f.x), f.y);
    float flux = raw * P.uMisc.y; // 定点 → 能量
    float cellArea = (poolSize.x * poolSize.y) / float(CAUSTICS_RES * CAUSTICS_RES);
    return vec3(flux / max(cellArea, 1e-8));
}


bool pointUnderwater(vec3 p) {
    return p.y < P.uSkyWater.y
        && all(greaterThan(p.xz, P.uSkyWater.zw))
        && all(lessThan(p.xz, P.uSkyWater.zw + P.uPoolWorld.xy));
}


// ---- 水下光臂（体积单散射）------------------------------------------------
// 太阳经水面折射聚焦后在水中形成 3D 能量场：以焦散光子图按 xz 外推，
// 并补偿"光子图已含到池底吸收"的部分（越浅越亮）→ 光臂与池底焦散
// 同源同图案，随波纹动画流动；水上水下任何视角都可见。
const float WATER_SCATTER = 0.15;    // 散射系数（略夸张的"果冻水"）
const float SCATTER_GAIN = 3.0;     // 前向散射增益（视觉调整）
vec3 causticsSunEnergy(vec3 p) {
    if (!pointUnderwater(p) || P.uFrame.w < 0.5) return vec3(0.0);
    vec3 E = causticsIrradiance(p) * P.uSunColor.rgb * P.uSun.w;
    // 光子溅射在池底(y≈1)，含全程吸收；采样点更浅 → 补回这段吸收
    return E * exp(WATER_SIGMA * max(p.y - 1.0, 0.0));
}

// ============================================================================
//  场景遍历：体素 DDA（含水介质状态机）+ 每段直线段内的球体测试。
//  球体在折射事件【之前】测试 → 水面上的球不会被水面错误遮挡。
//  glint 累积空气→介质界面的太阳镜面高光（期望值，主循环并入 radiance）
//  返回：0=天空 1=体素命中 2=球体命中
// ============================================================================
int traceScene(vec3 ro, inout vec3 rd, inout vec3 throughput, inout uint seed,
               inout vec3 glint,
               out vec3 hitPos, out vec3 nHit, out uint matHit, out int sphIdx) {
    ivec3 ws = worldSize();
    sphIdx = -1; matHit = 0u; nHit = vec3(0, 1, 0); hitPos = ro;

    float tEnter, tExit;
    bool insideWorld = worldEnter(ro, rd, tEnter, tExit);

    // 世界 AABB 之前的段（相机在世界外时）测球
    vec3 pos = ro;
    if (!insideWorld) {
        float tS; int si = nearestSphere(ro, rd, 1e5, tS);
        if (si >= 0) {
            hitPos = ro + rd * tS;
            nHit = normalize(hitPos - sphereData[si * 3].xyz);
            sphIdx = si;
            return 2;
        }
        return 0; // 世界也碰不到 → 天空
    }
    if (tEnter > 0.0) {
        float tS; int si = nearestSphere(ro, rd, tEnter, tS);
        if (si >= 0) {
            hitPos = ro + rd * tS;
            nHit = normalize(hitPos - sphereData[si * 3].xyz);
            sphIdx = si;
            return 2;
        }
    }

    pos = ro + rd * max(tEnter + 1e-5, 0.0);
    ivec3 startCell = ivec3(clamp(floor(pos), vec3(0.0), vec3(ws) - 1.0));
    uint startMat = voxelAt(startCell);

    // 关键修复：起始格若是固体 → 立即命中。
    // 否则相机在世界外（如下方仰视底面）时，光线会穿透整个第一层，
    // 命中第二层的内部分界面，把体素内部晶格暴露成"蜂窝网格"。
    if (startMat != MAT_EMPTY && !isMedium(startMat)) {
        float t0 = max(tEnter, 0.0);
        hitPos = ro + rd * t0;
        if (tEnter > 0.0) {
            // 从世界外进入：法线 = AABB 入射面法线
            vec3 invD0 = 1.0 / safeDir(rd);
            vec3 tA0 = (vec3(0.0) - ro) * invD0;
            vec3 tB0 = (vec3(ws) - ro) * invD0;
            vec3 tmn0 = min(tA0, tB0);
            float mx0 = max(max(tmn0.x, tmn0.y), tmn0.z);
            if (mx0 == tmn0.x)      nHit = vec3(-sign(rd.x), 0.0, 0.0);
            else if (mx0 == tmn0.y) nHit = vec3(0.0, -sign(rd.y), 0.0);
            else                    nHit = vec3(0.0, 0.0, -sign(rd.z));
        } else {
            nHit = -rd;   // 起点本就在固体内（防御性兜底，防透视）
        }
        matHit = startMat;
        return 1;
    }

    // 统一介质状态：0=空气；MAT_WATER=水；10..16=彩色玻璃
    uint medium = isMedium(startMat) ? startMat : 0u;
    float medDist = 0.0;
    float totalT = max(tEnter, 0.0);

    for (int restart = 0; restart < 8; restart++) {
        ivec3 cell = ivec3(clamp(floor(pos), vec3(0.0), vec3(ws) - 1.0));
        vec3 sd = safeDir(rd);
        vec3 deltaDist = abs(1.0 / sd);
        ivec3 stepDir = ivec3(sign(rd));
        vec3 nextBound = vec3(cell) + max(stepDir, ivec3(0));
        vec3 tMax = max((nextBound - pos) / sd, vec3(0.0));
        bool tir = false;

        for (int i = 0; i < 256; i++) {
            float tStep; vec3 n;
            if (tMax.x < tMax.y && tMax.x < tMax.z) {
                tStep = tMax.x; n = vec3(-float(stepDir.x), 0.0, 0.0);
                tMax.x += deltaDist.x; cell.x += stepDir.x;
            } else if (tMax.y < tMax.z) {
                tStep = tMax.y; n = vec3(0.0, -float(stepDir.y), 0.0);
                tMax.y += deltaDist.y; cell.y += stepDir.y;
            } else {
                tStep = tMax.z; n = vec3(0.0, 0.0, -float(stepDir.z));
                tMax.z += deltaDist.z; cell.z += stepDir.z;
            }

            // 本段 [totalT, tStep] 行进在【旧介质】里——先算出长度，
            // 任何结算/切换都必须把它记在旧介质账上（否则单层玻璃吸收记零）
            float dSeg = max(tStep - totalT, 0.0);

            // 水下光臂：水中每段中点采一次单散射（焦散能量场照明）
            if (medium == MAT_WATER && dSeg > 1e-4) {
                vec3 sp = pos + rd * (totalT + dSeg * 0.5);
                vec3 E = causticsSunEnergy(sp);
                // 视线到采样点的透射率 + 前向散射相位加成
                vec3 Tv = exp(-WATER_SIGMA * (medDist + dSeg * 0.5));
                float phase = (1.0 + SCATTER_GAIN * max(dot(rd, -P.uSun.xyz), 0.0)) / (4.0 * PI);
                glint += throughput * (WATER_SCATTER * phase) * E * Tv * dSeg;
            }

            // 出界：本段剩余部分测球后按天空处理
            if (any(lessThan(cell, ivec3(0))) || any(greaterThanEqual(cell, ws))) {
                float tS; int si = nearestSphere(pos, rd, 1e5, tS);
                if (si >= 0) {
                    if (medium != 0u) throughput *= exp(-mediumSigma(medium) * (medDist + min(tS, dSeg)));
                    hitPos = pos + rd * tS;
                    nHit = normalize(hitPos - sphereData[si * 3].xyz);
                    sphIdx = si;
                    return 2;
                }
                if (medium != 0u) throughput *= exp(-mediumSigma(medium) * (medDist + dSeg));
                return 0; // 天空
            }

            uint m = voxelAt(cell);
            if (isMedium(m)) {
                if (medium != m) {
                    // 介质切换事件：先测本段球体（玻璃/水面前方的球可见）
                    float tS; int si = nearestSphere(pos, rd, tStep, tS);
                    if (si >= 0) {
                        if (medium != 0u) throughput *= exp(-mediumSigma(medium) * (medDist + min(tS, dSeg)));
                        hitPos = pos + rd * tS;
                        nHit = normalize(hitPos - sphereData[si * 3].xyz);
                        sphIdx = si;
                        return 2;
                    }
                    // 界面法线（水面顶面用波纹法线）
                    vec3 bn = (medium == 0u && m == MAT_WATER && n.y > 0.5)
                            ? waterNormal((pos + rd * tStep).xz) : n;
                    // 精确菲涅尔：反射 vs 折射随机分支（TIR 时 F=1 → 必反射）
                    float etaI = (medium == 0u) ? 1.0 : mediumEta(medium);
                    float etaT = mediumEta(m);
                    if (etaI == etaT) {
                        // 同折射率界面（如相邻不同色玻璃）：不弯折不反射，
                        // 本段记旧介质账后切换——DDA 状态仍有效，无需重启
                        throughput *= exp(-mediumSigma(medium) * (medDist + dSeg));
                        medium = m;
                        medDist = 0.0;
                    } else {
                        float F = fresnelDielectric(dot(-rd, bn), etaI, etaT);
                        // 太阳镜面高光（期望值 F×反射亮度；天空已无日盘，不重复计数）
                        if (medium == 0u) glint += throughput * F * sunSpecular(rd, bn);
                        vec3 bpos = pos + rd * tStep;   // 界面点
                        if (rnd(seed) < F) {
                            // 镜面反射：留在原介质（本段继续累计，跨重启保留）
                            medDist += dSeg;
                            rd = reflect(rd, bn);
                            pos = bpos + rd * 1e-4;
                            totalT = 0.0;
                            tir = true;
                            break;
                        }
                        // 折射进入新介质：结算旧介质吸收（含本段）
                        throughput *= exp(-mediumSigma(medium) * (medDist + dSeg));
                        vec3 r = refract(rd, bn, etaI / etaT);
                        if (dot(r, r) < 1e-8) { medDist += dSeg; rd = reflect(rd, bn); pos = bpos + rd * 1e-4; totalT = 0.0; tir = true; break; }
                        // 折射后必须完整重启 DDA：方向变了，旧的 tMax 步进状态全部作废
                        rd = normalize(r);
                        medium = m;
                        medDist = 0.0;
                        pos = bpos + rd * 1e-4;
                        totalT = 0.0;
                        tir = true;
                        break;
                    }
                } else {
                    medDist += dSeg;   // 同介质继续行进
                }
            } else if (m != MAT_EMPTY) {
                // 固体命中：先测本段球体
                float tS; int si = nearestSphere(pos, rd, tStep, tS);
                if (si >= 0) {
                    if (medium != 0u) throughput *= exp(-mediumSigma(medium) * (medDist + min(tS, dSeg)));
                    hitPos = pos + rd * tS;
                    nHit = normalize(hitPos - sphereData[si * 3].xyz);
                    sphIdx = si;
                    return 2;
                }
                if (medium != 0u) throughput *= exp(-mediumSigma(medium) * (medDist + dSeg));
                hitPos = pos + rd * tStep;
                nHit = n; matHit = m;
                return 1;
            } else if (medium != 0u) {
                // 介质→空气：先测本段球体
                float tS; int si = nearestSphere(pos, rd, tStep, tS);
                if (si >= 0) {
                    throughput *= exp(-mediumSigma(medium) * (medDist + min(tS, dSeg)));
                    hitPos = pos + rd * tS;
                    nHit = normalize(hitPos - sphereData[si * 3].xyz);
                    sphIdx = si;
                    return 2;
                }
                // 出射界面法线：波纹几何法线朝上，但折射需要朝向入射侧（向下）
                // ——方向反了会让菲涅尔恒判 TIR，水下永远看不到水外（Snell window 消失）
                vec3 bn2 = (medium == MAT_WATER && n.y < -0.5)
                         ? -waterNormal((pos + rd * tStep).xz) : n;
                float F2 = fresnelDielectric(dot(-rd, bn2), mediumEta(medium), 1.0);
                vec3 bpos2 = pos + rd * tStep;
                if (rnd(seed) < F2) {
                    // 菲涅尔反射（含 TIR）：留在介质内重启（本段继续累计）
                    medDist += dSeg;
                    rd = reflect(rd, bn2);
                    pos = bpos2 + rd * 1e-4;
                    totalT = 0.0;
                    tir = true;
                    break;
                }
                throughput *= exp(-mediumSigma(medium) * (medDist + dSeg));
                vec3 r = refract(rd, bn2, mediumEta(medium));
                if (dot(r, r) < 1e-8) { medDist += dSeg; rd = reflect(rd, bn2); pos = bpos2 + rd * 1e-4; totalT = 0.0; tir = true; break; }
                // 折射出射后完整重启 DDA（同入射理由：tMax 已作废）
                rd = normalize(r);
                medium = 0u;
                medDist = 0.0;
                pos = bpos2 + rd * 1e-4;
                totalT = 0.0;
                tir = true;
                break;
            }
            totalT = tStep;
        }
        if (!tir) break;
    }
    return 0;
}

// ---- 阴影射线透过率：穿过水/玻璃按 Beer-Lambert 染色（彩色投影的来源）-----
// 遇不透明体素返回 vec3(0)；纯空气路径返回 vec3(1)。
vec3 shadowTransmittance(vec3 ro, vec3 rd, float dist) {
    float tEnter, tExit;
    if (!worldEnter(ro, rd, tEnter, tExit)) return vec3(1.0);
    if (tEnter > dist) return vec3(1.0);
    ivec3 ws = worldSize();
    vec3 pos = ro + rd * max(tEnter + 1e-4, 0.0);

    ivec3 cell = ivec3(clamp(floor(pos), vec3(0.0), vec3(ws) - 1.0));
    // 起始格已是固体 → 全遮挡
    uint m0 = voxelAt(cell);
    if (m0 != MAT_EMPTY && !isMedium(m0)) return vec3(0.0);
    uint medium = isMedium(m0) ? m0 : 0u;

    vec3 sd = safeDir(rd);
    vec3 deltaDist = abs(1.0 / sd);
    ivec3 stepDir = ivec3(sign(rd));
    vec3 nextBound = vec3(cell) + max(stepDir, ivec3(0));
    vec3 tMax = max((nextBound - pos) / sd, vec3(0.0));

    float dTarget = dist - length(pos - ro);   // 目标点在段参数系里的距离
    float seg = 0.0;
    vec3 T = vec3(1.0);

    for (int i = 0; i < 256; i++) {
        float tStep;
        if (tMax.x < tMax.y && tMax.x < tMax.z)      { tStep = tMax.x; tMax.x += deltaDist.x; cell.x += stepDir.x; }
        else if (tMax.y < tMax.z)                    { tStep = tMax.y; tMax.y += deltaDist.y; cell.y += stepDir.y; }
        else                                          { tStep = tMax.z; tMax.z += deltaDist.z; cell.z += stepDir.z; }
        if (tStep >= dTarget || tStep > 1e5) break;
        if (any(lessThan(cell, ivec3(0))) || any(greaterThanEqual(cell, ws))) break;

        uint m = voxelAt(cell);
        if (m != medium) {
            // 介质切换：结算上一段吸收
            T *= exp(-mediumSigma(medium) * max(tStep - seg, 0.0));
            if (m != MAT_EMPTY && !isMedium(m)) return vec3(0.0);   // 不透明
            medium = m;
            seg = tStep;
        }
    }
    T *= exp(-mediumSigma(medium) * max(dTarget - seg, 0.0));
    return T;
}

// ---- 采样工具 ------------------------------------------------------------
vec3 onb(vec3 n, vec3 d) {
    vec3 t = normalize(abs(n.y) < 0.99 ? cross(vec3(0, 1, 0), n) : cross(vec3(1, 0, 0), n));
    vec3 b = cross(n, t);
    return normalize(t * d.x + b * d.y + n * d.z);
}

// ---- PBR：GGX 微表面 ------------------------------------------------------
// 标准 GGX 半向量采样（Walter et al. 2007 公开算法）
vec3 sampleGGX(vec3 n, float rough, vec2 u) {
    float a = max(rough * rough, 1e-3);
    float phi = 2.0 * PI * u.x;
    float cosT = sqrt((1.0 - u.y) / (1.0 + (a * a - 1.0) * u.y));
    float sinT = sqrt(max(0.0, 1.0 - cosT * cosT));
    return onb(n, vec3(sinT * cos(phi), sinT * sin(phi), cosT));
}

// 程序化泳池瓷砖：砖面光亮 + 砖缝粗糙深色 + 每砖微色差
// 返回调制后的 (albedo.rgb, 粗糙度)
vec4 poolTilePattern(vec3 p, vec3 n) {
    // 面向 UV：取法线以外的两个坐标轴
    vec2 tuv = (abs(n.y) > 0.5) ? p.xz : (abs(n.x) > 0.5 ? p.zy : p.xy);
    const float TILE = 2.0;                 // 每米 2 块砖（0.5m 砖）
    vec2 f = fract(tuv * TILE);
    vec2 dEdge = min(f, 1.0 - f);
    float e = min(dEdge.x, dEdge.y);        // 到砖缝的距离（砖空间）
    float tileMask = smoothstep(0.0, 0.10, e);   // 0=缝 1=砖面
    // 每砖哈希色差
    vec2 tid = floor(tuv * TILE);
    uint h = uint(tid.x) * 1973u + uint(tid.y) * 9277u;
    pcgHash(h); pcgHash(h);
    float var = 0.92 + 0.14 * (float(h) / 4294967296.0);
    vec3 tileCol = vec3(0.60, 0.72, 0.74) * var;
    vec3 groutCol = vec3(0.30, 0.32, 0.33);
    vec3 col = mix(groutCol, tileCol, tileMask);
    float rough = mix(0.75, 0.16, tileMask);   // 砖缝粗糙、砖面光亮
    return vec4(col, rough);
}

// ---- 坏灯闪烁：按灯位置哈希决定故障模式（均值归一化到 1，不改变整体亮度）----
// 灯光时钟：移动模式取 0（闪烁均值=1，时间混合下确定平滑）；静止取真实时钟
float lampT() { return (P.uTemporal.x > 0.5) ? 0.0 : P.uExposure.z; }
float lightFlicker(vec3 cell, float t) {
    if (t <= 0.0) return 1.0;                       // 时钟冻结（A/B 测试）→ 全亮
    float h = hash13(cell);
    if (h < 0.55) return 1.0;                       // 多数灯正常
    float h2 = hash13(cell + vec3(7.1, 3.3, 11.7));
    if (h < 0.75) {
        // 频闪灯：均值 1.0（1.0 + 0.7·sin）
        return 1.0 + 0.7 * sin(t * (28.0 + 44.0 * h2) + h * 63.0);
    }
    // 抽搐坏灯：35% 时间掉到 0.05，其余 1.28+0.45·sin 补偿（均值 ≈ 1.0）
    float seg = floor(t * (1.5 + 3.0 * h2));
    float drop = hash13(vec3(seg, h2, 3.7));
    if (drop < 0.35) return 0.05;
    return 1.28 + 0.45 * sin(t * 37.0 + h * 40.0);
}

// ---- 后室材质的程序化纹理 -----------------------------------------------
// 壁纸(17)：竖条纹+污渍；地毯(18)：颗粒；天花板(19)：1m 格栅。返回 albedo 乘数
vec3 materialPattern(uint m, vec3 p, vec3 n, inout float rough) {
    if (m == 17u) {
        float u = abs(n.x) > 0.5 ? p.z : p.x;
        float stripe = 0.94 + 0.06 * sin(u * 12.56);
        float grunge = 0.92 + 0.08 * hash13(floor(p * 4.0));
        return vec3(stripe * grunge);
    }
    if (m == 18u) {
        return vec3(0.88 + 0.12 * hash13(floor(p * 6.0)));
    }
    if (m == 19u) {
        vec2 tuv = (abs(n.y) > 0.5) ? p.xz : (abs(n.x) > 0.5 ? p.zy : p.xy);
        vec2 f = fract(tuv);
        vec2 d = min(f, 1.0 - f);
        float grid = smoothstep(0.0, 0.05, min(d.x, d.y));
        rough = mix(0.6, 0.85, grid);
        return vec3(mix(0.45, 1.0, grid));
    }
    return vec3(1.0);
}
vec3 cosSample(vec3 n, vec2 u) {
    float r = sqrt(u.x);
    float phi = 2.0 * PI * u.y;
    return onb(n, vec3(r * cos(phi), r * sin(phi), sqrt(max(0.0, 1.0 - u.x))));
}
vec3 coneSample(vec3 dir, float cosMax, vec2 u) {
    float cz = 1.0 - u.x * (1.0 - cosMax);
    float sz = sqrt(max(0.0, 1.0 - cz * cz));
    float phi = 2.0 * PI * u.y;
    return onb(dir, vec3(sz * cos(phi), sz * sin(phi), cz));
}
float schlick(float f0, float cosT) { return f0 + (1.0 - f0) * pow(1.0 - cosT, 5.0); }

// ---- GGX 微表面 PBR（alb.w = 粗糙度，如泳池瓷砖）--------------------------
// 太阳方向的确定性 GGX 镜面评估（NoL 与分母约掉；返回辐亮度贡献）
vec3 ggxSunSpec(vec3 rd, vec3 n, float rough) {
    vec3 v = -rd;
    vec3 l = P.uSun.xyz;
    float NoL = dot(n, l);
    if (NoL <= 0.0) return vec3(0.0);
    vec3 h = normalize(v + l);
    float NoH = max(dot(n, h), 0.0);
    float NoV = max(dot(n, v), 1e-4);
    float VoH = max(dot(v, h), 0.0);
    float a = rough * rough;
    float a2 = a * a;
    float dd = NoH * NoH * (a2 - 1.0) + 1.0;
    float D = a2 / (PI * dd * dd + 1e-7);          // GGX 法线分布
    float k = a * 0.5;
    float G = (NoV / (NoV * (1.0 - k) + k))        // Smith 几何（Schlick-GGX 近似）
            * (NoL / (NoL * (1.0 - k) + k));
    float F = 0.04 + 0.96 * pow(1.0 - VoH, 5.0);   // 介电体菲涅尔
    return P.uSunColor.rgb * (P.uSun.w * D * G * F / (4.0 * NoV));
}
// GGX 半程向量重要性采样（Walter 2007）
vec3 ggxSampleHalf(vec3 n, float rough, vec2 u) {
    float a = rough * rough;
    float phi = 2.0 * PI * u.x;
    float ct = sqrt((1.0 - u.y) / (1.0 + (a * a - 1.0) * u.y));
    float st = sqrt(max(0.0, 1.0 - ct * ct));
    return onb(n, vec3(st * cos(phi), st * sin(phi), ct));
}

// ---- 天空：噪声 → 体积云 → 昼夜/星星 --------------------------------------
const float CLOUD_LOW = 42.0, CLOUD_HIGH = 78.0;   // 云层高度（世界之上）

// 三线性值噪声（hash13 已在文件顶部定义）
float vnoise(vec3 p) {
    vec3 i = floor(p), f = fract(p);
    f = f * f * (3.0 - 2.0 * f);
    float a = mix(mix(mix(hash13(i), hash13(i + vec3(1,0,0)), f.x),
                      mix(hash13(i + vec3(0,1,0)), hash13(i + vec3(1,1,0)), f.x), f.y),
                  mix(mix(hash13(i + vec3(0,0,1)), hash13(i + vec3(1,0,1)), f.x),
                      mix(hash13(i + vec3(0,1,1)), hash13(i + vec3(1,1,1)), f.x), f.y), f.z);
    return a;
}
float fbm3(vec3 p) {
    float v = 0.0, a = 0.55;
    for (int i = 0; i < 3; i++) {
        v += a * vnoise(p);
        p = p * 2.13 + vec3(17.3, 9.1, 23.7);
        a *= 0.5;
    }
    return v;
}

// 云密度：FBM 形状 + 高度包络 + 风移动（时间复用水面时钟）
float cloudDensity(vec3 p) {
    float h = (p.y - CLOUD_LOW) / (CLOUD_HIGH - CLOUD_LOW);
    if (h < 0.0 || h > 1.0) return 0.0;
    vec3 q = p * 0.016 + vec3(P.uMisc.z * 0.05, 0.0, P.uMisc.z * 0.021);
    float shape = fbm3(q);
    float env = 1.0 - abs(h - 0.45) * 1.7;
    return clamp((shape - 0.46) * 3.2 * clamp(env, 0.0, 1.0), 0.0, 1.0);
}

// 太阳线穿云透射（云影；3 采样近似）
float cloudShadow(vec3 p) {
    if (P.uSun.y < 0.03) return 1.0;
    float t0 = (CLOUD_LOW - p.y) / P.uSun.y;
    float dsum = 0.0;
    for (int i = 0; i < 3; i++) {
        dsum += cloudDensity(p + P.uSun.xyz * (t0 + float(i) * 14.0 + 7.0));
    }
    return exp(-dsum * 0.6);
}

// 星星：方向哈希点（夜空出现，被云遮挡参与混合）
float starField(vec3 rd) {
    vec3 p = rd * 260.0;
    vec3 c = floor(p);
    float h = hash13(c);
    if (h < 0.9965) return 0.0;
    vec3 f = fract(p) - 0.5;
    float d2 = dot(f, f);
    float mag = (h - 0.9965) / 0.0035;
    return smoothstep(0.18, 0.0, d2) * (0.4 + 1.8 * mag);
}

// 天空底色：昼夜过渡 + 黄昏暖色 + 星星
vec3 skyBase(vec3 rd) {
    float sunY = P.uSun.y;
    float dayF = smoothstep(-0.08, 0.28, sunY);
    float h = clamp(rd.y * 0.5 + 0.5, 0.0, 1.0);
    // 昼
    vec3 dayHor = vec3(0.72, 0.76, 0.82) * 0.45;
    vec3 dayZen = vec3(0.30, 0.42, 0.72) * 0.75;
    // 夜
    vec3 nightHor = vec3(0.040, 0.055, 0.10);
    vec3 nightZen = vec3(0.012, 0.020, 0.055);
    vec3 hor = mix(nightHor, dayHor, dayF);
    vec3 zen = mix(nightZen, dayZen, dayF);
    // 黄昏：太阳方位的地平线加暖
    float dusk = clamp(1.0 - abs(sunY - 0.06) / 0.22, 0.0, 1.0) * dayF;
    if (dusk > 0.001) {
        float az = max(dot(normalize(vec3(rd.x, 0.0, rd.z)),
                           normalize(vec3(P.uSun.x, 0.0, P.uSun.z))), 0.0);
        float prox = pow(az, 3.0) * pow(1.0 - clamp(rd.y, 0.0, 1.0), 2.5);
        hor = mix(hor, vec3(1.05, 0.42, 0.14), prox * dusk * 0.85);
    }
    vec3 base = mix(hor, zen, pow(h, 0.7)) * P.uSkyWater.x;
    // 星星（夜因子）
    float night = 1.0 - smoothstep(-0.06, 0.16, sunY);
    if (night > 0.001 && rd.y > 0.0)
        base += vec3(0.85, 0.9, 1.0) * starField(rd) * night * P.uSkyWater.x;
    return base;
}

// 体积云步进（16 步；梯度光照 + 前向散射；星星在云后）
// 无硬裁剪：4km 内都可步进，远处云雾化融入天空——地平线处平滑消失
vec3 skyColor(vec3 ro, vec3 rd) {
    vec3 base = skyBase(rd);
    if (rd.y < 0.002 || P.uSkyWater.x <= 0.001) return base;
    const float MAXD = 4000.0;                      // 最大步进距离
    float t0 = (CLOUD_LOW - ro.y) / rd.y;
    float t1 = (CLOUD_HIGH - ro.y) / rd.y;
    t1 = min(t1, t0 + MAXD);
    if (t0 > MAXD) return base;
    float stepLen = clamp((t1 - t0) / 16.0, 3.0, 60.0);
    float jitter = fract(float(P.uFrame.x) * 0.618034);
    vec3 sunC = P.uSunColor.rgb * P.uSun.w * smoothstep(-0.10, 0.12, P.uSun.y);
    float T = 1.0;
    vec3 acc = vec3(0.0);
    for (int i = 0; i < 16; i++) {
        float t = t0 + (float(i) + jitter) * stepLen;
        if (t > t1) break;
        vec3 p = ro + rd * t;
        float d = cloudDensity(p);
        if (d > 0.008) {
            // 朝太阳的密度差 → 自阴影
            float dSun = cloudDensity(p + P.uSun.xyz * 13.0);
            float light = exp(-max(dSun - d, 0.0) * 4.5) * 0.85 + 0.15;
            float ph = 0.6 + 0.9 * max(dot(rd, P.uSun.xyz), 0.0);   // 前向散射
            vec3 li = sunC * light * ph + base * 1.1;
            float a = 1.0 - exp(-d * stepLen * 0.16);
            // 距离雾化：1.5km 起渐隐，4km 完全融入天空（无切边）
            a *= smoothstep(MAXD, 1500.0, t);
            if (a <= 0.001) continue;
            acc += T * a * li;
            T *= 1.0 - a;
            if (T < 0.05) break;
        }
    }
    return acc + base * T;
}

// ---- 直接光 NEE（阴影走光谱透过率 → 彩色投影；太阳乘云影）-----------------
vec3 evalDirect(vec3 p, vec3 n, bool uw, inout uint seed) {
    vec3 E = vec3(0.0);
    if (P.uFrame.w > 0.5) {
        if (uw) {
            // 水下走焦散光子图（光子发射端已乘云影）
            E += causticsIrradiance(p) * P.uSunColor.rgb * P.uSun.w;
        } else {
            vec3 L = P.uSun.xyz;
            float NoL = dot(n, L);
            if (NoL > 0.0) {
                vec3 so = p + n * 1e-3;
                vec3 T = shadowTransmittance(so, L, 1e5) * sphereShadowTint(so, L, 1e5);
                E += P.uSunColor.rgb * P.uSun.w * NoL * T * cloudShadow(so);
            }
        }
    }
    int nS = int(P.uMisc.w);
    for (int i = 0; i < nS; i++) {
        vec4 em = sphereData[i * 3 + 2];
        if (dot(em.rgb, em.rgb) < 1e-6) continue;
        vec4 s = sphereData[i * 3];
        vec3 toC = s.xyz - p;
        float d = length(toC);
        if (d < s.w + 1e-4) continue;
        float sinMax = s.w / d;
        float cosMax = sqrt(max(0.0, 1.0 - sinMax * sinMax));
        // 解析立体角 + 中心方向余弦（漫反射零方差估计，替代锥方向采样——
        // 小面积光偏差可忽略，直接光噪点大幅下降）
        vec3 wi = toC / d;
        float NoL = dot(n, wi);
        if (NoL <= 0.0) continue;
        vec3 so = p + n * 1e-3;
        vec3 T = shadowTransmittance(so, wi, d - s.w) * sphereShadowTint(so, wi, d - s.w);
        if (dot(T, T) < 1e-6) continue;
        float solid = 2.0 * PI * (1.0 - cosMax);
        float fl = lightFlicker(floor(s.xyz), lampT());   // 坏灯闪烁
        E += em.rgb * (NoL * solid) * T * fl;
    }
    // 发光体素（灯板/霓虹）：等效球面解析 NEE——零方差面光源
    int nE = int(P.uBake.x);
    for (int i = 0; i < nE; i++)
    {
        vec3 c = emissiveData[i * 2].xyz;
        vec3 em = emissiveData[i * 2 + 1].rgb;
        vec3 toC = c - p;
        float d = length(toC);
        if (d < 0.9) continue;                     // 光源自身/紧邻表面跳过
        vec3 wi = toC / d;
        float NoL = dot(n, wi);
        if (NoL <= 0.0) continue;
        // 等效半径 0.45（≈1m² 发光面），与球灯同一解析式。
        // 阴影射线必须停在光源体素之外（体素半对角 ~0.87，取 0.95 防自遮挡）
        float sinMax = 0.45 / d;
        float cosMax = sqrt(max(0.0, 1.0 - sinMax * sinMax));
        float solid = 2.0 * PI * (1.0 - cosMax);
        vec3 so = p + n * 1e-3;
        vec3 T = shadowTransmittance(so, wi, d - 0.95);
        if (dot(T, T) < 1e-6) continue;
        float fl = lightFlicker(floor(c), lampT());
        E += em * (NoL * solid) * T * fl;
    }
    return E;
}
