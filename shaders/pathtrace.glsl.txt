#version 450
// ============================================================================
//  pathtrace.glsl — 体素路径追踪（渐进式累积）
//  原创实现，采用公开的经典算法：
//   - Amanatides & Woo (1987) 体素 DDA 遍历
//   - Beer-Lambert 介质吸收 / Snell 折射 / Schlick 菲涅尔近似
//   - 下一次事件估计 (NEE) + 余弦加权半球采样
//   - R2 低差异序列（像素抖动 + 时间累积抗锯齿）
// ============================================================================

layout(local_size_x = 16, local_size_y = 16) in;

// ---- 绑定 ----------------------------------------------------------------
layout(set = 0, binding = 0, std140) uniform Params {
    vec4 uCamPos;    // 相机位置
    vec4 uCamRight;  // right * tan(fov/2) * aspect
    vec4 uCamUp;     // up * tan(fov/2)
    vec4 uCamFwd;    // fwd * tan(fov/2)
    vec4 uRes;       // w, h, 1/w, 1/h
    vec4 uFrame;     // sampleIdx, randSeed, maxBounces, sunOn
    vec4 uSun;       // dir.xyz(指向太阳), intensity
    vec4 uSunColor;  // rgb, exposure
    vec4 uSkyWater;  // skyIntensity, waterSurfaceY, poolMinX, poolMinZ
    vec4 uPoolWorld; // poolSizeX, poolSizeZ, worldSX, worldSY
    vec4 uMisc;      // worldSZ, causticsDecode, time, sphereCount
    vec4 uSelect;    // （present 用）
    vec4 uExposure;  // x=autoOn, y=key, z=灯光时钟（坏灯闪烁）, w=GI缓存开关
    vec4 uTemporal;  // x=移动模式, y=混合alpha, z=spp, w=无历史首帧
    vec4 uPrevPos; vec4 uPrevRight; vec4 uPrevUp; vec4 uPrevFwd;
    vec4 uBake;      // x=发光体素数（灯板解析 NEE）
} P;

layout(set = 0, binding = 1, std430) readonly buffer VoxelBuf { uint voxelWords[]; };
layout(set = 0, binding = 2, std430) readonly buffer SphereBuf { vec4 sphereData[]; };  // 每球 3 个 vec4
layout(set = 0, binding = 3, std430) readonly buffer MatBuf    { vec4 matData[]; };     // 每材质 2 个 vec4
layout(set = 0, binding = 4, rgba16f) uniform image2D accumImg;
layout(set = 0, binding = 5, r32ui) readonly uniform uimage2D causticsImg;

// ---- GI 辐照缓存（借鉴原版 directional GI 思路）---------------------------
// 16³ 探针覆盖世界（每 2 体素一个），存"总辐照 = 直接 + 间接"，
// 时间混合逐帧收敛 → 漫反射间接光平滑无噪；多弹射以辐射度迭代方式收敛
layout(set = 0, binding = 6) uniform sampler3D giCacheSmp;   // SamplerWithTexture
layout(set = 0, binding = 7, rgba16f) uniform image3D giCacheImg;
// 时间重投影模式：本帧辐射与主命中位置写到独立纹理（temporal.glsl 混合）
layout(set = 0, binding = 8, rgba16f) uniform image2D frameImg;
layout(set = 0, binding = 9, rgba16f) uniform image2D posImg;
// 发光体素列表（每项 2 个 vec4：中心 / 发光色）——灯板解析 NEE
layout(set = 0, binding = 10, std430) readonly buffer EmissiveBuf { vec4 emissiveData[]; };
// 三角网格（车等完整模型）：顶点/均匀网格/材质
layout(set = 0, binding = 11, std430) readonly buffer MeshVertBuf { vec4 meshVerts[]; };
layout(set = 0, binding = 12, std430) readonly buffer MeshGridTriBuf { uvec4 meshGridTris[]; };
layout(set = 0, binding = 13, std430) readonly buffer MeshGridInfoBuf { vec4 meshGridInfo[]; };
layout(set = 0, binding = 14, std430) readonly buffer MeshGridCellBuf { vec4 meshGridCells[]; };
layout(set = 0, binding = 15, std430) readonly buffer MeshMatBuf  { vec4 meshMats[]; };

#include "common.h.txt"

// ---- 三角网格求交（均匀网格 DDA + Möller-Trumbore）-------------------------
bool traceMesh(vec3 ro, vec3 rd, float tmax, out float tHit, out vec3 nHit, out uint matHit) {
    vec4 info = meshGridInfo[0];  // minX minY minZ cellSize
    const int G = 16;
    vec3 gMin = info.xyz;
    float cell = info.w;
    vec3 gMax = gMin + cell * float(G);

    // AABB 粗筛
    vec3 invD = 1.0 / safeDir(rd);
    vec3 ta = (gMin - ro) * invD;
    vec3 tb = (gMax - ro) * invD;
    vec3 tmn = min(ta, tb), tmx = max(ta, tb);
    float tEnter = max(max(tmn.x, tmn.y), tmn.z);
    float tExit = min(min(tmx.x, tmx.y), tmx.z);
    if (tEnter > tExit || tExit < 0.0 || tEnter > tmax) return false;

    // DDA 进入点
    vec3 pos = ro + rd * max(tEnter + 1e-4, 0.0);
    ivec3 cellI = clamp(ivec3((pos - gMin) / cell), ivec3(0), ivec3(G - 1));
    vec3 sd = safeDir(rd);
    vec3 deltaD = abs(1.0 / sd);
    ivec3 stepD = ivec3(sign(rd));
    vec3 nextB = vec3(cellI) + max(stepD, ivec3(0));
    vec3 tMax = max((nextB * cell + gMin - pos) / sd, 0.0);

    float bestT = tmax;
    int bestTri = -1;
    float curT = max(tEnter + 1e-4, 0.0);

    for (int step = 0; step < 96; step++) {
        if (any(lessThan(cellI, ivec3(0))) || any(greaterThanEqual(cellI, ivec3(G)))) break;
        if (curT > bestT) break;

        // 当前格子的三角形
        int ci = cellI.x + cellI.y * G + cellI.z * G * G;
        vec2 fc = meshGridCells[ci].xy;  // firstTri, count
        int first = int(fc.x);
        int cnt = int(fc.y);
        for (int i = 0; i < cnt; i++) {
            int ti = first + i;
            uvec4 tri = meshGridTris[ti];
            vec3 v0 = meshVerts[int(tri.x)].xyz;
            vec3 v1 = meshVerts[int(tri.y)].xyz;
            vec3 v2 = meshVerts[int(tri.z)].xyz;
            vec3 e1 = v1 - v0;
            vec3 e2 = v2 - v0;
            vec3 pv = cross(rd, e2);
            float det = dot(e1, pv);
            if (abs(det) < 1e-10) continue;
            float invDet = 1.0 / det;
            vec3 tv = ro - v0;
            float u = dot(tv, pv) * invDet;
            if (u < 0.0 || u > 1.0) continue;
            vec3 qv = cross(tv, e1);
            float v = dot(rd, qv) * invDet;
            if (v < 0.0 || u + v > 1.0) continue;
            float tt = dot(e2, qv) * invDet;
            if (tt > 1e-4 && tt < bestT) { bestT = tt; bestTri = ti; }
        }

        // DDA 步进
        float tStep;
        if (tMax.x < tMax.y && tMax.x < tMax.z) {
            tStep = tMax.x; tMax.x += deltaD.x; cellI.x += stepD.x;
        } else if (tMax.y < tMax.z) {
            tStep = tMax.y; tMax.y += deltaD.y; cellI.y += stepD.y;
        } else {
            tStep = tMax.z; tMax.z += deltaD.z; cellI.z += stepD.z;
        }
        curT = tStep;
    }

    if (bestTri >= 0) {
        tHit = bestT;
        uvec4 tri = meshGridTris[bestTri];
        vec3 v0 = meshVerts[int(tri.x)].xyz;
        vec3 v1 = meshVerts[int(tri.y)].xyz;
        vec3 v2 = meshVerts[int(tri.z)].xyz;
        nHit = normalize(cross(v1 - v0, v2 - v0));
        if (dot(nHit, rd) > 0.0) nHit = -nHit;
        matHit = tri.w;
        return true;
    }
    return false;
}

void main() {
    ivec2 pix = ivec2(gl_GlobalInvocationID.xy);
    if (pix.x >= int(P.uRes.x) || pix.y >= int(P.uRes.y)) return;

    uint seed = uint(pix.x) * 1973u + uint(pix.y) * 9277u
              + uint(P.uFrame.y) * 26699u + uint(P.uFrame.x) * 1259u;
    // 两轮哈希：单轮 PCG 对线性相关的相邻种子去相关不足，
    // 会让相邻像素产生帧间相关的噪声（表现为稳定的 2px 网格）
    pcgHash(seed);
    pcgHash(seed);

    // R2 低差异抖动（逐像素计数，空间+时间都均匀）
    vec2 r2 = vec2(0.7548776662466927, 0.5698402909980532);
    float r2n = float(uint(P.uFrame.x) * 4194304113u
                    + uint(pix.x) * 2246822519u + uint(pix.y) * 3266489917u);
    vec2 jitter = fract(r2 * r2n) - 0.5;
    vec2 uv = (vec2(pix) + 0.5 + jitter) * P.uRes.zw * 2.0 - 1.0;
    uv.y = -uv.y;   // 纹理第 0 行 = 屏幕顶部 → 应朝上看

    vec3 rd = normalize(P.uCamFwd.xyz + uv.x * P.uCamRight.xyz + uv.y * P.uCamUp.xyz);
    vec3 ro = P.uCamPos.xyz;

    vec3 radiance = vec3(0.0);
    vec3 throughput = vec3(1.0);
    bool lastSpecular = true;
    vec3 firstHit = ro + rd * 400.0;   // 主命中位置（时间重投影用；天空=远点）
    bool haveFirst = false;
    vec3 rd0 = rd;                     // 初始光线（检测介质折射弯折）
    bool pathSpecular = false;         // 主路径含镜面/折射 → temporal 提高混合权重

    int maxB = int(P.uFrame.z);
    for (int bounce = 0; bounce < maxB; bounce++) {
        vec3 p, n; uint mVox; int si;
        vec3 glint = vec3(0.0);
        int kind = traceScene(ro, rd, throughput, seed, glint, p, n, mVox, si);

        // 正常网格求交
        {
            float mT; vec3 mN; uint mMat;
            if (traceMesh(ro, rd, 1e5, mT, mN, mMat)) {
                float segT = (kind != 0) ? length(p - ro) : 1e5;
                if (mT < segT) {
                    kind = 3;
                    p = ro + rd * mT;
                    n = mN;
                    mVox = mMat;
                    si = -1;
                }
            }
        }

        // 三角网格（车）：如果比体素/球更近则替换命中
        {
            float mT; vec3 mN; uint mMat;
            if (traceMesh(ro, rd, 1e5, mT, mN, mMat)) {
                float segT = (kind != 0) ? length(p - ro) : 1e5;
                if (mT < segT) {
                    kind = 3;
                    p = ro + rd * mT;
                    n = mN;
                    mVox = mMat;
                    si = -1;
                }
            }
        }

        radiance += glint;

        if (!haveFirst) {
            firstHit = (kind != 0) ? p : ro + rd * 400.0;
            haveFirst = true;
            // 首段被水/玻璃折射弯折 → 像素内容是折射景象（随视角变化快）
            if (dot(rd - rd0, rd - rd0) > 1e-6) pathSpecular = true;
        }

        if (kind == 0) {
            radiance += throughput * skyColor(ro, rd);
            break;
        }

        if (kind == 2) {
            // ================= 球体 =================
            vec4 s = sphereData[si * 3];
            vec4 alb = sphereData[si * 3 + 1];
            vec4 em  = sphereData[si * 3 + 2];

            if (alb.w == SMAT_GLASS) {
                float cosI = clamp(-dot(rd, n), 0.0, 1.0);
                float Fg = schlick(GLASS_F0, cosI);
                radiance += throughput * Fg * sunSpecular(rd, n);   // 入射面太阳高光（期望值）
                if (rnd(seed) < Fg) {
                    rd = reflect(rd, n); ro = p + rd * 1e-4;
                    lastSpecular = true; if (bounce == 0) pathSpecular = true; continue;
                }
                vec3 rin = refract(rd, n, 1.0 / 1.5);
                if (dot(rin, rin) < 1e-8) { rd = reflect(rd, n); ro = p + rd * 1e-4; continue; }
                vec3 pos = p + rin * 1e-4;
                vec3 dir = normalize(rin);
                for (int k = 0; k < 4; k++) {
                    float tEx = max(sphereExitT(pos, dir, s), 1e-4);
                    pos += dir * tEx;
                    vec3 nEx = normalize(pos - s.xyz);
                    throughput *= exp(-alb.rgb * tEx);           // Beer-Lambert
                    vec3 rEx = refract(dir, nEx, 1.5);
                    float fExit = schlick(GLASS_F0, clamp(dot(dir, nEx), 0.0, 1.0));
                    if (dot(rEx, rEx) > 1e-8 && rnd(seed) >= fExit) {
                        rd = normalize(rEx); ro = pos + rd * 1e-4;
                        break;
                    }
                    dir = reflect(dir, nEx);
                    pos += dir * 1e-4;
                    if (k == 3) { rd = dir; ro = pos; }
                }
                lastSpecular = true; if (bounce == 0) pathSpecular = true;
                continue;
            }
            if (alb.w == SMAT_METAL) {
                radiance += throughput * alb.rgb * sunSpecular(rd, n);   // 金属太阳高光
                vec3 fuzz = (vec3(rnd(seed), rnd(seed), rnd(seed)) - 0.5) * 0.05;
                rd = normalize(reflect(rd, n) + fuzz);
                ro = p + n * 1e-4;
                throughput *= alb.rgb;
                lastSpecular = true; if (bounce == 0) pathSpecular = true;
                continue;
            }
            if (dot(em.rgb, em.rgb) > 1e-6) {
                // 灯球本体：直接加自发光（×坏灯闪烁）后结束
                radiance += throughput * em.rgb
                          * lightFlicker(floor(sphereData[si * 3].xyz), lampT());
                break;
            }
            // ---- 漫反射：NEE 直接光 + 单段弹射读 GI 缓存（平滑无噪）----
            bool uw = pointUnderwater(p);
            vec3 Edir = evalDirect(p, n, uw, seed);
            radiance += throughput * alb.rgb * Edir / PI;
            vec3 rdB = cosSample(n, rnd2(seed));
            vec3 tpB = vec3(1.0);
            vec3 pq, nq; uint mq; int sq;
            vec3 glintB = vec3(0.0);
            if (P.uExposure.w > 0.5) {
                // ---- 模式 A：单段弹射读 GI 缓存 ----
                int kq = traceScene(p + n * 1e-3, rdB, tpB, seed, glintB, pq, nq, mq, sq);
                vec3 Lq;
                if (kq == 0) {
                    Lq = skyColor(p + n * 1e-3, rdB);
                } else if (kq == 2 && dot(sphereData[sq * 3 + 2].rgb,
                                          sphereData[sq * 3 + 2].rgb) > 1e-6) {
                    Lq = vec3(0.0);   // 灯光贡献由 NEE 负责（避免双重计数 + 椒盐噪点）
                } else if (kq == 1 && matEmission(mq).w == MTYPE_EMISSIVE) {
                    Lq = vec3(0.0);
                } else {
                    vec4 albq = (kq == 2) ? sphereData[sq * 3 + 1] : matAlbedo(mq);
                    Lq = albq.rgb * cacheE(pq) / PI;
                }
                radiance += throughput * alb.rgb * (Lq * tpB + glintB);
                cacheSplat(p, Edir + Lq * PI, 0.05);
                break;
            }
            // ---- 模式 B：经典多弹射路径追踪 ----
            rd = rdB;
            ro = p + n * 1e-3;
            throughput *= alb.rgb;
            lastSpecular = false;
        } else if (kind == 3) {
            // ================= 三角网格（完整模型）=========================
            vec4 alb = meshMats[int(mVox) * 2];       // albedo + roughness
            vec4 em  = meshMats[int(mVox) * 2 + 1];   // emission + type
            float rough = clamp(alb.w, 0.02, 1.0);

            if (em.w == 2.0) {   // 发光（车灯）
                radiance += throughput * em.rgb;
                break;
            }
            if (em.w == 1.0) {   // 金属（镀铬）
                radiance += throughput * alb.rgb * sunSpecular(rd, n);
                vec3 fuzz = (vec3(rnd(seed), rnd(seed), rnd(seed)) - 0.5) * rough;
                rd = normalize(reflect(rd, n) + fuzz);
                ro = p + n * 1e-4;
                throughput *= alb.rgb;
                lastSpecular = true;
                continue;
            }
            if (em.w == 3.0) {   // 玻璃（车窗：单面薄折射）
                float cosI = clamp(-dot(rd, n), 0.0, 1.0);
                float F = 0.04 + 0.96 * pow(1.0 - cosI, 5.0);
                if (rnd(seed) < F) {
                    rd = reflect(rd, n); ro = p + n * 1e-4;
                    lastSpecular = true;
                    continue;
                }
                vec3 rin = refract(rd, n, 1.0 / 1.5);
                if (dot(rin, rin) > 1e-8) {
                    throughput *= exp(-alb.rgb * 0.12);  // 薄玻璃吸收
                    rd = normalize(rin);
                    ro = p + rd * 2e-3;
                    lastSpecular = true;
                    continue;
                }
                rd = reflect(rd, n); ro = p + n * 1e-4;
                lastSpecular = true;
                continue;
            }
            // ---- 漫反射 / 车漆（GGX 光泽 + NEE + GI 缓存弹射）----
            {
                bool uw = pointUnderwater(p);
                vec3 Edir = evalDirect(p, n, uw, seed);
                radiance += throughput * alb.rgb * Edir / PI;

                // 电介质菲涅尔镜面瓣（车漆光泽反射）
                float cosV = clamp(dot(n, -rd), 0.0, 1.0);
                float F = 0.04 + 0.96 * pow(1.0 - cosV, 5.0);
                if (P.uFrame.w > 0.5)
                    radiance += throughput * F * exp(-(rough * rough) * 40.0) * sunSpecular(rd, n);
                if (rough < 0.95 && rnd(seed) < F) {
                    vec3 wh = sampleGGX(n, rough, rnd2(seed));
                    vec3 dir = reflect(rd, wh);
                    if (dot(dir, n) > 0.0) {
                        rd = dir; ro = p + n * 1e-4;
                        lastSpecular = true;
                        if (bounce == 0) pathSpecular = true;
                        continue;
                    }
                }
                // GI 单段弹射
                vec3 rdB = cosSample(n, rnd2(seed));
                vec3 tpB = vec3(1.0);
                vec3 pq, nq; uint mq; int sq;
                vec3 glintB = vec3(0.0);
                int kq = traceScene(p + n * 1e-3, rdB, tpB, seed, glintB, pq, nq, mq, sq);
                vec3 Lq;
                if (kq == 0) {
                    Lq = skyColor(p + n * 1e-3, rdB);
                } else if (kq == 2 && dot(sphereData[sq * 3 + 2].rgb,
                                          sphereData[sq * 3 + 2].rgb) > 1e-6) {
                    Lq = vec3(0.0);
                } else if (kq == 1 && matEmission(mq).w == MTYPE_EMISSIVE) {
                    Lq = vec3(0.0);
                } else if (kq == 3) {
                    vec4 albq = meshMats[int(mq) * 2];
                    Lq = albq.rgb * cacheE(pq) / PI;
                } else {
                    vec4 albq = (kq == 2) ? sphereData[sq * 3 + 1] : matAlbedo(mq);
                    Lq = albq.rgb * cacheE(pq) / PI;
                }
                radiance += throughput * alb.rgb * (Lq * tpB + glintB);
                cacheSplat(p, Edir + Lq * PI, 0.05);
                break;
            }
        } else {
            // ================= 体素（PBR：粗糙度驱动的镜面/漫反射双瓣）=========
            vec4 alb = matAlbedo(mVox);
            vec4 em = matEmission(mVox);
            float rough = clamp(alb.w, 0.02, 1.0);

            if (em.w == MTYPE_EMISSIVE) {
                // 发光体素 × 坏灯闪烁（按所在格哈希故障模式）
                float fl = lightFlicker(floor(p - n * 0.01), lampT());
                radiance += throughput * em.rgb * fl;
                break;
            }
            if (em.w == MTYPE_METAL) {
                radiance += throughput * alb.rgb * sunSpecular(rd, n);   // 金属太阳高光
                vec3 fuzz = (vec3(rnd(seed), rnd(seed), rnd(seed)) - 0.5) * alb.w;
                rd = normalize(reflect(rd, n) + fuzz);
                ro = p + n * 1e-4;
                throughput *= alb.rgb;
                lastSpecular = true; if (bounce == 0) pathSpecular = true;
                continue;
            }

            // 泳池瓷砖：程序化砖缝纹理调制颜色与粗糙度
            if (mVox == MAT_POOLTILE) {
                vec4 pat = poolTilePattern(p, n);
                alb.rgb = pat.rgb;
                rough = pat.w;
            }
            // 后室材质：壁纸/地毯/天花板的程序化纹理
            if (mVox >= 17u && mVox <= 19u) {
                alb.rgb *= materialPattern(mVox, p, n, rough);
            }

            // 电介质菲涅尔（F0=0.04）决定镜面瓣概率
            float cosV = clamp(dot(n, -rd), 0.0, 1.0);
            float F = 0.04 + 0.96 * pow(1.0 - cosV, 5.0);
            // 太阳高光：近镜面瓣的期望贡献（随粗糙度衰减）
            if (P.uFrame.w > 0.5)
                radiance += throughput * F * exp(-(rough * rough) * 40.0) * sunSpecular(rd, n);

            if (rough < 0.95 && rnd(seed) < F) {
                // ---- GGX 镜面瓣 ----
                vec3 wh = sampleGGX(n, rough, rnd2(seed));
                vec3 dir = reflect(rd, wh);
                if (dot(dir, n) > 0.0) {
                    rd = dir;
                    ro = p + n * 1e-4;
                    lastSpecular = true; if (bounce == 0) pathSpecular = true;
                    continue;
                }
                // 反射方向朝下 → 退回漫反射
            }
            // ---- 漫反射瓣：NEE 直接光 + 单段弹射读 GI 缓存（平滑无噪）----
            bool uw = pointUnderwater(p);
            vec3 Edir = evalDirect(p, n, uw, seed);
            radiance += throughput * alb.rgb * Edir / PI;
            vec3 rdB = cosSample(n, rnd2(seed));
            vec3 tpB = vec3(1.0);
            vec3 pq, nq; uint mq; int sq;
            vec3 glintB = vec3(0.0);
            if (P.uExposure.w > 0.5) {
                // ---- 模式 A：单段弹射读 GI 缓存（平滑无噪，原版思路）----
                int kq = traceScene(p + n * 1e-3, rdB, tpB, seed, glintB, pq, nq, mq, sq);
                vec3 Lq;
                if (kq == 0) {
                    Lq = skyColor(p + n * 1e-3, rdB);
                } else if (kq == 2 && dot(sphereData[sq * 3 + 2].rgb,
                                          sphereData[sq * 3 + 2].rgb) > 1e-6) {
                    Lq = vec3(0.0);   // 灯光贡献由 NEE 负责（避免双重计数 + 椒盐噪点）
                } else if (kq == 1 && matEmission(mq).w == MTYPE_EMISSIVE) {
                    Lq = vec3(0.0);
                } else {
                    vec4 albq = (kq == 2) ? sphereData[sq * 3 + 1] : matAlbedo(mq);
                    Lq = albq.rgb * cacheE(pq) / PI;
                }
                radiance += throughput * alb.rgb * (Lq * tpB + glintB);
                cacheSplat(p, Edir + Lq * PI, 0.05);
                break;
            }
            // ---- 模式 B：经典多弹射路径追踪（对照/高精度）----
            rd = rdB;
            ro = p + n * 1e-3;
            throughput *= alb.rgb;
            lastSpecular = false;
        }

        if (bounce > 2) {
            float q = clamp(max(throughput.r, max(throughput.g, throughput.b)), 0.02, 0.95);
            if (rnd(seed) > q) break;
            throughput /= q;
        }
    }

    if (any(isnan(radiance)) || any(isinf(radiance))) radiance = vec3(0.0);
    // 萤火虫钳制：抑制罕见高亮采样主导方差（降噪关键一环）
    radiance = min(radiance, vec3(40.0));

    if (P.uTemporal.x > 0.5) {
        // 移动模式：本帧样本（累加 spp 次）+ 主命中位置 → temporal 混合
        imageStore(frameImg, pix, vec4(radiance, 1.0));
        imageStore(posImg, pix, vec4(firstHit, pathSpecular ? 1.0 : 0.0));
    } else {
        // 静止模式：纯累积（收敛最优）
        vec4 prev = imageLoad(accumImg, pix);
        vec4 outVal = (P.uFrame.x < 0.5) ? vec4(radiance, 1.0) : prev + vec4(radiance, 1.0);
        imageStore(accumImg, pix, outVal);
    }
}
