#version 450
// 清空焦散光子图（场景修改后调用）
layout(local_size_x = 16, local_size_y = 16) in;
layout(set = 0, binding = 0, r32ui) uniform uimage2D causticsImg;

void main() {
    ivec2 pix = ivec2(gl_GlobalInvocationID.xy);
    if (pix.x >= 256 || pix.y >= 256) return;
    imageStore(causticsImg, pix, uvec4(0u));
}
