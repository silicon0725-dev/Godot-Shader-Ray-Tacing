#version 450
// ============================================================================
//  caustics.glsl — 水体焦散光子映射
//  从太阳向水面发射光子网格（256×256/帧，跨帧累积收敛）：
//   入射 → Snell 折射进入水体 → Beer-Lambert 衰减 → DDA 到达池底 →
//   双线性定点数 atomicAdd 溅射到 r32ui 光子图
// ============================================================================

layout(local_size_x = 16, local_size_y = 16) in;

layout(set = 0, binding = 0, std140) uniform Params {
    vec4 uCamPos;
    vec4 uCamRight;
    vec4 uCamUp;
    vec4 uCamFwd;
    vec4 uRes;
    vec4 uFrame;     // x=sampleIdx, y=randSeed, z=maxBounces, w=sunOn
    vec4 uSun;       // xyz=指向太阳, w=intensity
    vec4 uSunColor;
    vec4 uSkyWater;  // x=skyIntensity, y=waterSurfaceY, z=poolMinX, w=poolMinZ
    vec4 uPoolWorld; // x=poolSizeX, y=poolSizeZ, z=worldSX, w=worldSY
    vec4 uMisc;      // x=worldSZ, y=causticsDecode, z=time, w=sphereCount
} P;

layout(set = 0, binding = 1, std430) readonly buffer VoxelBuf { uint voxelWords[]; };
layout(set = 0, binding = 2, r32ui) uniform uimage2D causticsImg;

const uint MAT_EMPTY = 0u;
const uint MAT_WATER = 6u;
const uint MAT_GLASS_MIN = 10u;
const uint MAT_GLASS_MAX = 16u;
bool isGlassVox(uint m) { return m >= MAT_GLASS_MIN && m <= MAT_GLASS_MAX; }
const vec3 WATER_SIGMA = vec3(0.55, 0.18, 0.10);
const float WATER_ETA = 1.33;
const uint CAUSTICS_RES = 256u;
const uint PHOTON_GRID = 512u;   // 每帧每方向光子数
const float FIXED_SCALE = 4096.0; // 定点数编码比例

ivec3 worldSize() { return ivec3(int(P.uPoolWorld.z), int(P.uPoolWorld.w), int(P.uMisc.x)); }
vec3 safeDir(vec3 d) { return mix(d, vec3(1e-10), lessThan(abs(d), vec3(1e-10))); }

uint pcgHash(inout uint s) {
    s = s * 747796405u + 2891336453u;
    uint w = ((s >> ((s >> 28u) + 4u)) ^ s) * 277803737u;
    return (w >> 22u) ^ w;
}
float rnd(inout uint s) { return float(pcgHash(s)) * (1.0 / 4294967296.0); }
vec2  rnd2(inout uint s) { return vec2(rnd(s), rnd(s)); }

uint voxelAt(ivec3 c) {
    ivec3 ws = worldSize();
    if (any(lessThan(c, ivec3(0))) || any(greaterThanEqual(c, ws))) return MAT_EMPTY;
    int idx = c.x + c.y * ws.x + c.z * ws.x * ws.y;
    return (voxelWords[idx >> 2] >> ((idx & 3) * 8)) & 0xFFu;
}

// 与 pathtrace.glsl 一致的水面波纹法线（迭代分形波 + 时间流动）
vec3 waterNormal(vec2 p) {
    float t = P.uMisc.z;
    vec2 pos = p * 0.55;
    float iter = 0.0;
    float frequency = 2.1;
    float speed = 1.3;
    float weight = 1.0;
    float totalW = 0.0;
    vec2 grad = vec2(0.0);
    for (int i = 0; i < 5; i++) {
        vec2 dir = vec2(sin(iter), cos(iter));
        float phase = dot(dir, pos) * frequency + t * speed;
        float w = exp(sin(phase) - 1.0);
        grad += w * cos(phase) * weight * dir * frequency;
        iter += 1.9;
        weight *= 0.72;
        frequency *= 1.55;
        speed *= 1.12;
        totalW += weight;
    }
    float amp = 0.16 / max(totalW, 1e-3);
    return normalize(vec3(-grad.x * amp, 1.0, -grad.y * amp));
}

// ---- 云影（与 pathtrace.glsl 同款噪声/云层，光子权重乘云透射）----
const float CLOUD_LOW = 42.0, CLOUD_HIGH = 78.0;
float hash13(vec3 p) {
    p = fract(p * 0.1031);
    p += dot(p, p.zyx + 31.32);
    return fract((p.x + p.y) * p.z);
}
float vnoise(vec3 p) {
    vec3 i = floor(p), f = fract(p);
    f = f * f * (3.0 - 2.0 * f);
    float a = mix(mix(mix(hash13(i), hash13(i + vec3(1,0,0)), f.x),
                      mix(hash13(i + vec3(0,1,0)), hash13(i + vec3(1,1,0)), f.x), f.y),
                  mix(mix(hash13(i + vec3(0,0,1)), hash13(i + vec3(1,0,1)), f.x),
                      mix(hash13(i + vec3(0,1,1)), hash13(i + vec3(1,1,1)), f.x), f.y), f.z);
    return a;
}
float fbm3(vec3 p) {
    float v = 0.0, a = 0.55;
    for (int i = 0; i < 3; i++) {
        v += a * vnoise(p);
        p = p * 2.13 + vec3(17.3, 9.1, 23.7);
        a *= 0.5;
    }
    return v;
}
float cloudDensity(vec3 p) {
    float h = (p.y - CLOUD_LOW) / (CLOUD_HIGH - CLOUD_LOW);
    if (h < 0.0 || h > 1.0) return 0.0;
    vec3 q = p * 0.016 + vec3(P.uMisc.z * 0.05, 0.0, P.uMisc.z * 0.021);
    float shape = fbm3(q);
    float env = 1.0 - abs(h - 0.45) * 1.7;
    return clamp((shape - 0.46) * 3.2 * clamp(env, 0.0, 1.0), 0.0, 1.0);
}
float cloudShadow(vec3 p) {
    if (P.uSun.y < 0.03) return 1.0;
    float t0 = (CLOUD_LOW - p.y) / P.uSun.y;
    float dsum = 0.0;
    for (int i = 0; i < 3; i++) {
        dsum += cloudDensity(p + P.uSun.xyz * (t0 + float(i) * 14.0 + 7.0));
    }
    return exp(-dsum * 0.6);
}

// 光子穿过水体（忽略水与空气边界）DDA 到固体或出界；返回行进距离，未命中返回 -1
float photonMarch(vec3 ro, vec3 rd) {
    ivec3 ws = worldSize();
    vec3 sd = safeDir(rd);
    vec3 invD = 1.0 / sd;
    vec3 tA = (vec3(0.0) - ro) * invD;
    vec3 tB = (vec3(ws) - ro) * invD;
    vec3 mn = min(tA, tB), mx = max(tA, tB);
    float tEnter = max(max(mn.x, mn.y), mn.z);
    float tExit  = min(min(mx.x, mx.y), mx.z);
    if (tEnter > tExit || tExit < 0.0) return -1.0;

    vec3 pos = ro + rd * max(tEnter + 1e-4, 0.0);
    ivec3 cell = ivec3(clamp(floor(pos), vec3(0.0), vec3(ws) - 1.0));
    vec3 deltaDist = abs(invD);
    ivec3 stepDir = ivec3(sign(rd));
    vec3 nextBound = vec3(cell) + max(stepDir, ivec3(0));
    vec3 tMax = max((nextBound - pos) / sd, vec3(0.0));

    for (int i = 0; i < 256; i++) {
        float tStep;
        if (tMax.x < tMax.y && tMax.x < tMax.z)      { tStep = tMax.x; tMax.x += deltaDist.x; cell.x += stepDir.x; }
        else if (tMax.y < tMax.z)                    { tStep = tMax.y; tMax.y += deltaDist.y; cell.y += stepDir.y; }
        else                                          { tStep = tMax.z; tMax.z += deltaDist.z; cell.z += stepDir.z; }
        if (any(lessThan(cell, ivec3(0))) || any(greaterThanEqual(cell, ws))) return -1.0;
        uint m = voxelAt(cell);
        if (m != MAT_EMPTY && m != MAT_WATER && !isGlassVox(m)) return tStep;
    }
    return -1;
}

// 定点双线性溅射
void splat(vec2 cellCoord, float weight) {
    vec2 g = cellCoord - 0.5;
    vec2 f = fract(g);
    ivec2 c0 = ivec2(floor(g));
    float w00 = (1.0 - f.x) * (1.0 - f.y);
    float w10 = f.x * (1.0 - f.y);
    float w01 = (1.0 - f.x) * f.y;
    float w11 = f.x * f.y;
    ivec2 mx = ivec2(int(CAUSTICS_RES) - 1);
    if (w00 > 0.0) imageAtomicAdd(causticsImg, clamp(c0, ivec2(0), mx), uint(w00 * weight * FIXED_SCALE + 0.5));
    if (w10 > 0.0) imageAtomicAdd(causticsImg, clamp(c0 + ivec2(1, 0), ivec2(0), mx), uint(w10 * weight * FIXED_SCALE + 0.5));
    if (w01 > 0.0) imageAtomicAdd(causticsImg, clamp(c0 + ivec2(0, 1), ivec2(0), mx), uint(w01 * weight * FIXED_SCALE + 0.5));
    if (w11 > 0.0) imageAtomicAdd(causticsImg, clamp(c0 + ivec2(1, 1), ivec2(0), mx), uint(w11 * weight * FIXED_SCALE + 0.5));
}

void main() {
    ivec2 gid = ivec2(gl_GlobalInvocationID.xy);
    if (gid.x >= int(PHOTON_GRID) || gid.y >= int(PHOTON_GRID)) return;
    if (P.uFrame.w < 0.5) return; // 太阳关闭时不发射

    uint seed = uint(gid.x) * 1973u + uint(gid.y) * 9277u + uint(P.uFrame.y) * 26699u;
    pcgHash(seed);
    pcgHash(seed);   // 二次哈希去相关（同 pathtrace 的理由）

    vec2 poolMin = P.uSkyWater.zw, poolSize = P.uPoolWorld.xy;
    float waterY = P.uSkyWater.y;

    // 水面上的抖动入射点
    vec2 u = (vec2(gid) + 0.5 + (rnd2(seed) - 0.5)) / float(PHOTON_GRID);
    vec2 xPos = poolMin + u * poolSize;

    // 太阳方向锥抖动（约 1° 角半径 → 柔和焦散纹路）
    // uSun 指向太阳，光子传播方向为 -sunDir 附近的锥体
    vec3 sunDir = normalize(P.uSun.xyz);
    float cz = 1.0 - rnd(seed) * (1.0 - cos(0.02));
    float sz = sqrt(max(0.0, 1.0 - cz * cz));
    float phi = 6.2831853 * rnd(seed);
    vec3 tan1 = normalize(cross(vec3(0, 1, 0), sunDir));
    vec3 bitan = cross(sunDir, tan1);
    vec3 dir = normalize(-(tan1 * (sz * cos(phi)) + bitan * (sz * sin(phi)) + sunDir * cz));

    if (dir.y >= -1e-4) return;

    // 与水面 y=waterY 求交（从水面上方 5m 处出发）
    vec3 ro = vec3(xPos.x, waterY + 5.0, xPos.y);
    float tSurf = (waterY - ro.y) / dir.y;
    vec3 pSurf = ro + dir * tSurf;

    // 折射进入水体（波纹法线 → 光线聚焦形成焦散）
    vec3 wn = waterNormal(xPos);
    vec3 refr = refract(dir, wn, 1.0 / WATER_ETA);
    if (dot(refr, refr) < 1e-8) return;

    // 折射角带来的能量通量折减（菲涅尔近似，掠射减弱）
    float cosI = clamp(dot(-dir, wn), 0.0, 1.0);
    float F = 0.02 + 0.98 * pow(1.0 - cosI, 5.0);

    // DDA 到池底（忽略水内部折射弯曲——已按折射方向直线行进）
    float t = photonMarch(pSurf + refr * 1e-4, refr);
    if (t < 0.0) return;

    vec3 pHit = pSurf + refr * t;
    // 溅射位置（限定池内）
    vec2 cellCoord = (pHit.xz - poolMin) / poolSize * float(CAUSTICS_RES);
    if (cellCoord.x < 0.0 || cellCoord.y < 0.0 ||
        cellCoord.x > float(CAUSTICS_RES) - 1.0 || cellCoord.y > float(CAUSTICS_RES) - 1.0) return;

    // 每光子能量权重：E·A·cosθ·(1-F)·e^{-σd}·云透射 / N
    float poolArea = poolSize.x * poolSize.y;
    float N = float(PHOTON_GRID * PHOTON_GRID);
    vec3 atten = exp(-WATER_SIGMA * t);
    float monoW = (1.0 - F) * (atten.r * 0.299 + atten.g * 0.587 + atten.b * 0.114);
    float weight = poolArea * cosI * monoW * cloudShadow(pSurf) / N;

    splat(cellCoord, weight);
}
